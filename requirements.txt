# FraudGuard runtime dependencies (deployed Vercel backend).
#
# These are the pinned dependencies required to start the FastAPI app and serve
# `/`, `/health`, `/predict`, `/flagged` and `/metrics-lite`, including the model
# at `artifacts/models/fraud_model.ubj`.
#
# The deployed artifact is the trained XGBoost Booster in XGBoost's native UBJ
# format (exported from the Phase 3 Joblib bundle by
# `scripts/export_native_model.py`). Reading it needs only `xgboost` and `numpy`:
# the decision threshold, feature order and training metadata are stored inside
# the file as Booster attributes. Consequently joblib, scikit-learn and pandas are
# NOT runtime dependencies, and the deployed image never unpickles anything.
#
# `scipy` is not listed because `xgboost` declares it and imports it at import
# time; pip installs it automatically. Do not add pandas/scikit-learn here unless
# the loader starts reading the legacy Joblib bundle in production.
#
# Every version below is pinned to the set verified locally on Windows + CPython
# 3.13 (all install from prebuilt cp313 wheels; no compiler required).
#
# Local development dependencies — notebooks, EDA, tests, load testing, DB tooling
# and the Joblib fallback — live in `requirements-dev.txt`. Vercel installs
# `requirements.txt` only.

# --- API framework ---
fastapi==0.162.0
uvicorn[standard]==0.54.0

# --- Configuration + schema validation ---
pydantic==2.14.2
python-dotenv==1.2.4

# --- Database (SQLAlchemy ORM + psycopg 3 for the pooled Neon connection) ---
# Required by the deployed routes that access the alert database (/flagged,
# persist_alert). Excluding these would break the current API contract.
sqlalchemy==2.1.8
psycopg[binary]==3.3.8

# --- Inference stack (native XGBoost artifact only) ---
#   * numpy   — float64 feature matrix
#   * xgboost — Booster.load_model + DMatrix prediction (pulls scipy)
numpy==2.5.4
xgboost==3.4.1
